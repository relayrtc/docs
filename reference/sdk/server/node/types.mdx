## Import

```ts
import type { CreateRoomInput, ParticipantPermission, Room } from '@relayrtc/node'

const input: CreateRoomInput = { name: 'Team call', maxParticipants: 10 }
const permissions: readonly ParticipantPermission[] = ['room:join', 'audio:publish']

function roomLabel(room: Room) {
  return room.name
}
```

Types are exported from the package root. Validation rules and defaults are listed on each method page. A type describes the shape; the API still validates the request.

## JsonValue


## UsageResponse

Scoped usage totals, buckets and data quality. See [Usage schema](/reference/api/v1/schemas/usage).

```ts
export interface UsageResponse {
  readonly scope: { readonly organizationId: string; readonly projectId: string; readonly environmentId: string }
  readonly category: UsageCategory
  readonly window: {
    readonly range: UsageRange | 'custom'
    readonly startedAt: string
    readonly endedAt: string
    readonly timezone: 'UTC'
    readonly endExclusive: true
  }
  readonly bucketDefinition: {
    readonly granularity: 'hour' | 'day'
    readonly alignment: 'UTC calendar boundaries'
    readonly partialEdges: 'clipped to requested window'
    readonly durations: 'lifecycle intervals clipped to each bucket'
    readonly events: 'attributed to occurrence time'
  }
  readonly metricDefinitions: Readonly<Partial<Record<UsageMetric, UsageMetricDefinition>>>
  /** Included for all/streaming reports; service measurements are unpriced origin observations. */
  readonly serviceUsage?: ServiceUsageReport
  readonly summary: UsageMetrics
  readonly buckets: readonly { readonly startedAt: string; readonly endedAt: string; readonly metrics: UsageMetrics }[]
  readonly dataQuality: {
    readonly sessionHistory: 'complete' | 'partial'
    readonly messageHistory: 'complete' | 'partial'
    readonly turnTraffic: 'unavailable' | 'partial' | 'authoritative'
    readonly mediaDurations: 'observation_time'
  }
}
```

## ServiceUsageReport

Matching SDK builds export `ServiceUsageReport` and its metric, coverage, definition, resource and project types. `UsageResponse.serviceUsage` is optional because RTC-only categories do not return this report. See [service metrics and coverage](/reference/api/v1/schemas/usage#streaming-service-report).

```ts
export interface ServiceUsageReport {
  readonly window: {
    readonly startedAt: string
    readonly endedAt: string
    readonly timezone: 'UTC'
    readonly endExclusive: true
    readonly granularity: 'hour' | 'day'
  }
  readonly summary: ServiceUsageMetrics
  readonly metricDefinitions: Readonly<Record<ServiceUsageMetric, ServiceUsageMetricDefinition>>
  readonly rules: Readonly<Record<string, string>>
  readonly buckets: readonly {
    readonly startedAt: string
    readonly endedAt: string
    readonly metrics: ServiceUsageMetrics
  }[]
  readonly coverage: ServiceUsageMetricCoverage
  readonly journalStartedAt: string
  readonly historicalCoverage: 'complete' | 'partial'
  readonly deliveryAuthority: 'origin'
  readonly pricing: 'unpriced'
  readonly resources: readonly ServiceUsageResource[]
  readonly projects: readonly ServiceUsageProject[]
  readonly resourceLimit: 100
  readonly resourcesTruncated: boolean
}
```

`ServiceUsageCoverage` is `'observed' | 'estimated' | 'partial' | 'unavailable'`. Service metric values are separate from RTC `UsageMetrics`. They describe observed service quantities with `pricing: 'unpriced'`, not plan allowances or computed charges.
